cmake_minimum_required(VERSION 3.20)
project(vamos_engine LANGUAGES CXX)
add_executable(vamos_engine engine/main.cpp)
target_compile_features(vamos_engine PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(vamos_engine PRIVATE /W4)
else()
  target_compile_options(vamos_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME verifier COMMAND vamos_engine)

add_executable(vamos_transition_memory engine/transition_memory.cpp)
target_compile_features(vamos_transition_memory PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(vamos_transition_memory PRIVATE /W4)
else()
  target_compile_options(vamos_transition_memory PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME transition_memory COMMAND vamos_transition_memory)

option(VAMOS_METRO_FUSION "Build experimental SHA-256 route verifier" OFF)
if(VAMOS_METRO_FUSION)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  add_executable(vamos_metro_fusion engine/metro_fusion.cpp)
  target_compile_features(vamos_metro_fusion PRIVATE cxx_std_20)
  target_link_libraries(vamos_metro_fusion PRIVATE OpenSSL::Crypto)
  if(MSVC)
    target_compile_options(vamos_metro_fusion PRIVATE /W4)
  else()
    target_compile_options(vamos_metro_fusion PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME metro_fusion COMMAND vamos_metro_fusion)
endif()

add_executable(vamos_financial_guardian engine/financial_guardian.cpp)
target_compile_features(vamos_financial_guardian PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(vamos_financial_guardian PRIVATE /W4)
else()
  target_compile_options(vamos_financial_guardian PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME financial_guardian COMMAND vamos_financial_guardian)
